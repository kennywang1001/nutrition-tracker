# 營養標示的手機辨識（OCR）Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 新增食物的「拍營養標示」先在手機上辨識文字（Tesseract.js，不用額度、照片不離開裝置），文字送新的 `POST /api/foods/parse-label-text` 用確定性的規則讀出每 100 的營養素；讀不完整才「改用 AI 讀（用 1 次額度）」。另外可以貼上原況文字。

**Architecture:** 後端多一個純函式模組 `app/label_text.py`（正規化 → 找列 → 分欄 → 每一份量 → 選欄），換算與合理性檢查跟 `app/ai/label.py` 共用 `app/nutrition_units.py`（新的 `plausible_per_100`）；端點在 foods router、每人每分鐘 30 次、不碰資料庫、不寫 log。前端：`lib/label-ocr.ts`（主程式：縮圖、`import()`）＋`lib/label-ocr-engine.ts`（延遲載入：Tesseract.js 7.0.0、`chi_tra`、PSM 6、`rotateAuto`）；worker 與 core 用 `?url`、語言資料用 `vite.config.ts` 的小 plugin 放在 `/ocr/tessdata-4.0.0_best_int/`；拍營養標示的整段流程從 `NewFood` 搬進新元件 `LabelReader`，用 `onFill` 交回表單。

**Tech Stack:** FastAPI · Pydantic v2 · pytest · React 19 · TypeScript strict · TanStack Query v5 · Vite 8 · vite-plugin-pwa 1.3（workbox 7.4）· Tesseract.js 7.0.0 · Vitest · Playwright 1.63 · Biome

**依據規格：** `docs/superpowers/specs/2026-10-23-label-ocr-design.md`（以下稱「規格」；D1…D20 指它的 §2）

```python
"""從計畫抽出 `<!-- file: 路徑 -->` 底下的那一個 code block，原樣寫進 repo。

用法（repo 根目錄）：python -I $S/ocr-extract.py docs/superpowers/plans/2026-10-23-label-ocr.md <路徑...>
"""

import re
import sys
from pathlib import Path

plan = Path(sys.argv[1]).read_text(encoding="utf-8")
blocks = dict(re.findall(r"<!-- file: (\S+) -->\n```[a-z]*\n(.*?)\n```\n", plan, flags=re.S))
for path in sys.argv[2:]:
    target = Path(path)
    target.parent.mkdir(parents=True, exist_ok=True)
    target.write_text(blocks[path] + "\n", encoding="utf-8", newline="\n")
    print("wrote", path)
```

```python
"""掃控制字元與看不見的字元（Cc 除了 \n、Cf、Zl、Zp、BOM），有就以非零結束。

用法：python -I ocr-scan.py <檔案...>
"""

import sys
import unicodedata

sys.stdout.reconfigure(encoding="utf-8")
bad = 0
for path in sys.argv[1:]:
    with open(path, encoding="utf-8", newline="") as handle:
        text = handle.read()
    for line_no, line in enumerate(text.split("\n"), start=1):
        for col, char in enumerate(line, start=1):
            category = unicodedata.category(char)
            if char == "\t":
                continue
            if category in ("Cc", "Cf", "Zl", "Zp") or char == chr(0xFEFF):
                print(f"{path}:{line_no}:{col}: U+{ord(char):04X} {category}")
                bad += 1
    if "\r" in text:
        print(f"{path}: has CR")
        bad += 1
print(f"findings: {bad}")
sys.exit(1 if bad else 0)
```

```python
def plausible_per_100(
    base_unit: BaseUnit, kcal: Decimal, protein_g: Decimal, fat_g: Decimal, carb_g: Decimal
) -> NutritionInput | None:
    """外部讀來的每 100（AI 讀的標示、手機辨識或貼上的文字）能不能用：`NutritionInput` 的範圍
    （跟新增食物同一份定義，掃條碼規格 D27）＋物理上限（`is_physically_plausible`，審查 M3）。
    過不了 → None。兩條路共用這一份（標示 OCR 規格 D8）。"""
    try:
        checked = NutritionInput(
            base_unit=base_unit, kcal=kcal, protein_g=protein_g, fat_g=fat_g, carb_g=carb_g
        )
    except ValidationError:
        return None
    if not is_physically_plausible(checked.kcal, checked.protein_g, checked.fat_g, checked.carb_g):
        return None
    return checked
```

```python
_BORDERS = re.compile(r"[|\[\]]")
_SPACE_BEFORE_POINT = re.compile(r"(?<=\d)[ \t]+(?=\.\d)")
_SPACE_AFTER_POINT = re.compile(r"(?<=\d\.)[ \t]+(?=\d)")
_THOUSANDS = re.compile(r"(?<=\d),(?=\d{3}(?!\d))")
_DECIMAL_COMMA = re.compile(r"(?<=\d),(?=\d{1,2}(?!\d))")
_NUMBERISH = re.compile(r"[0-9OoIl.]+")
_LOOKALIKES = str.maketrans({"O": "0", "o": "0", "I": "1", "l": "1"})


def _fix_numberish(match: re.Match[str]) -> str:
    run = match.group()
    # 沒有真的數字的片段是字（Oil、Information），不動。
    if not any("0" <= char <= "9" for char in run):
        return run
    return run.translate(_LOOKALIKES)


def normalize(text: str) -> str:
    """全形 → 半形、表格線 → 空白、小數點兩邊的空白、逗號、數字裡的 O／l。少了小數點不猜。"""
    text = unicodedata.normalize("NFKC", text)  # ｜ 也在這裡變成 |
    text = _BORDERS.sub(" ", text)
    text = _SPACE_BEFORE_POINT.sub("", text)
    text = _SPACE_AFTER_POINT.sub("", text)
    text = _THOUSANDS.sub("", text)
    text = _DECIMAL_COMMA.sub(".", text)
    return _NUMBERISH.sub(_fix_numberish, text)

)
_ENERGY_IN_KCAL = re.compile(r"大卡|千卡|kcal", re.IGNORECASE)
_ENERGY_IN_KJ = re.compile(r"千焦|kj", re.IGNORECASE)


def _nutrients_in(line: str) -> list[Nutrient]:
    found: list[Nutrient] = []
    for nutrient, pattern in _KEYWORDS.items():
        if not pattern.search(line):
            continue
        if nutrient == "fat_g" and _FAT_QUALIFIERS.search(line):
            continue
        found.append(nutrient)
    return found


def _is_row_label(line: str) -> bool:
    return (
        bool(_nutrients_in(line))
        or bool(_OTHER_ROWS.search(line))
        or bool(_FAT_QUALIFIERS.search(line))
    )


@dataclass(frozen=True)
class _Row:
    nutrient: Nutrient
    label: str  # 第一個數字之前的字（`熱量(大卡)` 的單位在這裡）
    tokens: list[_Token]


def _rows(lines: list[str]) -> list[_Row]:
    """營養素的每一個候選列，照出現的順序。"""
    rows: list[_Row] = []
    for index, line in enumerate(lines):
        nutrients = _nutrients_in(line)
        # 兩種營養素在同一行（OCR 把兩列併在一起、或每日參考值的說明）→ 不用。
        if len(nutrients) != 1 or _OTHER_ROWS.search(line):
            continue
        first_number = _VALUE.search(line)
        label = line if first_number is None else line[: first_number.start()]
        tokens = _tokens(line)
        if not tokens and index + 1 < len(lines) and not _is_row_label(lines[index + 1]):
            # 名稱與數字在相鄰兩行（規格 §4.3）。
            tokens = _tokens(lines[index + 1])
        if tokens:
            rows.append(_Row(nutrients[0], label, tokens))
    return rows


# ---- 欄（規格 §4.2）----

_HEADER_SERVING = re.compile(r"每份|每一份(?!量)|per\s*serving", re.IGNORECASE)
_HEADER_PER_100 = re.compile(r"每\s*100\s*(公克|克|毫升|g|ml)|per\s*100\s*(g|ml)", re.IGNORECASE)
_HEADER_DV = re.compile(r"參考值|%\s*dv|daily\s*value", re.IGNORECASE)
_STANDARD = [Column.SERVING, Column.PER_100]  # 衛福部的格式：每份、每 100


def _plain_values(row: _Row) -> list[Decimal]:
    """不是熱量的那三種：克或沒單位（讀錯的單位也是沒單位）；% 與 mg、ml 不要。"""
    return [token.value for token in row.tokens if token.unit in ("g", None)]


def _columns(text: str, rows: list[_Row]) -> list[Column] | None:
    serving = _HEADER_SERVING.search(text)
    per_100 = _HEADER_PER_100.search(text)
    if serving and per_100:
        return _STANDARD if serving.start() < per_100.start() else [Column.PER_100, Column.SERVING]
    dv = _HEADER_DV.search(text) is not None
    # 只讀到一個（或零個）表頭：至少兩列剛好兩個數字、而且沒有每日參考值那一欄，才照衛福部的
    # 順序當成「每份、每 100」。有每日參考值時第二個數字可能是讀丟了 % 的百分比，不能猜。
    two_values = sum(1 for row in rows if row.nutrient != "kcal" and len(_plain_values(row)) == 2)
    if not dv and two_values >= 2:
        return _STANDARD
    if serving:
        return [Column.SERVING]
    if per_100:
        return [Column.PER_100]
    return None


def _fit(values: list[Decimal], width: int, dv: bool) -> list[Decimal] | None:
    if len(values) == width:
        return values
    if len(values) > width and dv:
        return values[:width]  # 每日參考值那一欄讀丟了 %：它在最後面
    return None


def _energy(row: _Row, width: int, dv: bool) -> list[Decimal] | None:
    kcal = [token.value for token in row.tokens if token.unit == "kcal"]
    kj = [token.value for token in row.tokens if token.unit == "kj"]
    if len(kcal) == width:
        return kcal
    if len(kj) == width:
        return [kj_to_kcal(value) for value in kj]
    bare = [token.value for token in row.tokens if token.unit is None]
    if _ENERGY_IN_KCAL.search(row.label):
        return _fit(bare, width, dv)
    if _ENERGY_IN_KJ.search(row.label):
        fitted = _fit(bare, width, dv)
        return None if fitted is None else [kj_to_kcal(value) for value in fitted]
    return None  # 熱量沒有單位：不猜是 kcal 還是 kJ


def _read_columns(
    rows: list[_Row], columns: list[Column], dv: bool
) -> dict[Column, dict[Nutrient, Decimal]]:
    result: dict[Column, dict[Nutrient, Decimal]] = {column: {} for column in columns}
    done: set[Nutrient] = set()
    for row in rows:
        if row.nutrient in done:
            continue  # 同一種有好幾行：用第一行分得出欄的
        if row.nutrient == "kcal":
            values = _energy(row, len(columns), dv)
        else:
            values = _fit(_plain_values(row), len(columns), dv)
        if values is None:
            continue
        done.add(row.nutrient)
        for column, value in zip(columns, values, strict=True):
            result[column][row.nutrient] = round_cents(value)
    return result


# ---- 每一份量、份數、單位（規格 §4.4）----

_SERVING_SIZE = re.compile(
    r"(?:每一?份量?|serving\s*size)\s*:?\s*(\d+(?:\.\d+)?)\s*(公克|克|g|毫升|ml)", re.IGNORECASE
)
_SERVINGS = re.compile(
    r"本包.{0,3}?(\d+(?:\.\d+)?)\s*份"
    r"|servings?\s*per\s*(?:package|container|pack)\s*:?\s*(?:about\s*)?(\d+(?:\.\d+)?)",
    re.IGNORECASE,
)


def _unit_of(word: str) -> BaseUnit:
    return BaseUnit.ML if word.lower() in ("毫升", "ml") else BaseUnit.G


def _serving_size(text: str) -> tuple[Decimal, BaseUnit] | None:
    match = _SERVING_SIZE.search(text)
    if match is None:
        return None
    size = round_cents(Decimal(match.group(1)))
    if not Decimal(0) < size <= MAX_SERVING_SIZE:
        return None
    return size, _unit_of(match.group(2))


def _servings(text: str) -> Decimal | None:
    match = _SERVINGS.search(text)
    if match is None:
        return None
    count = round_cents(Decimal(match.group(1) or match.group(2)))
    return count if Decimal(0) < count <= MAX_SERVINGS else None


# ---- 結果 ----


@dataclass(frozen=True)
class LabelText:
    """`found` 是**每 100**（g 或 ml，看 `base_unit`），讀不到的是 None；`complete` 時四個都有、
    驗過範圍與物理上限。`serving_size` 只在它的單位跟 `base_unit` 一樣時才有。"""

    complete: bool
    base_unit: BaseUnit
    found: Mapping[Nutrient, Decimal | None]
    missing: tuple[Nutrient, ...]
    reason: Reason | None
    basis: Basis | None
    serving_size: Decimal | None
    servings_per_package: Decimal | None


def _sane(nutrient: Nutrient, value: Decimal) -> bool:
    limit = MAX_KCAL_PER_100 if nutrient == "kcal" else MAX_MACRO_PER_100
    return value <= limit


def parse_label_text(text: str) -> LabelText:
    """營養標示的文字 → `LabelText`。純函式；任何輸入都不丟例外。"""
    normalized = normalize(text)
    lines = [line.strip() for line in normalized.split("\n") if line.strip()]
    rows = _rows(lines)
    columns = _columns(normalized, rows)
    dv = _HEADER_DV.search(normalized) is not None
    read = _read_columns(rows, columns, dv) if columns is not None else {}

    serving = _serving_size(normalized)
    per_100_header = _HEADER_PER_100.search(normalized)
    if per_100_header is not None:
        base_unit = _unit_of(per_100_header.group(1) or per_100_header.group(2))
    elif serving is not None:
        base_unit = serving[1]
    else:
        base_unit = BaseUnit.G
    size = serving[0] if serving is not None and serving[1] == base_unit else None
    servings = _servings(normalized)

    per_100 = read.get(Column.PER_100, {})
    per_serving = read.get(Column.SERVING, {})
    converted = (
        {nutrient: per_100_from_serving(value, size) for nutrient, value in per_serving.items()}
        if size is not None
        else {}
    )

    def result(
        found: Mapping[Nutrient, Decimal | None], reason: Reason | None, basis: Basis | None
    ) -> LabelText:
        missing = tuple(nutrient for nutrient in NUTRIENTS if found.get(nutrient) is None)
        return LabelText(
            complete=reason is None,
            base_unit=base_unit,
            found={nutrient: found.get(nutrient) for nutrient in NUTRIENTS},
            missing=missing,
            reason=reason,
            basis=basis,
            serving_size=size,
            servings_per_package=servings,
        )

    # 選欄（規格 D9）：完整而且合理的每 100 → 完整而且合理的每份（換算過）→ 不完整。
    implausible = False
    candidates: list[tuple[Basis, dict[Nutrient, Decimal]]] = [
        ("per_100", per_100),
        ("per_serving", converted),
    ]
    for basis, values in candidates:
        if len(values) != len(NUTRIENTS):
            continue
        checked = plausible_per_100(
            base_unit, values["kcal"], values["protein_g"], values["fat_g"], values["carb_g"]
        )
        if checked is None:
            implausible = True
            continue
        return result(
            {
                "kcal": checked.kcal,
                "protein_g": checked.protein_g,
                "fat_g": checked.fat_g,
                "carb_g": checked.carb_g,
            },
            None,
            basis,
        )

    if implausible:
        return result({}, "implausible", None)
    if columns is None:
        return result({}, "columns_unknown" if rows else "no_values", None)
    if not per_100 and not per_serving:
        return result({}, "no_values", None)
    if len(per_serving) > len(per_100) and size is None:
        return result({}, "needs_serving_size", None)
    best = per_100 if len(per_100) >= len(converted) else converted
    sane = {nutrient: value for nutrient, value in best.items() if _sane(nutrient, value)}
    return result(sane, "missing_fields", None)
```

```ts
import type { LabelReadResponse } from "../api/ai";
import type { LabelTextParse } from "../api/label-text";
import type { components } from "../api/schema";
import { formatMacro } from "./decimal";

type BaseUnit = components["schemas"]["BaseUnit"];
type Nutrient = LabelTextParse["missing"][number];

/** 要填進新增食物表單的東西（`NewFood` 的 `applyLabelFill`）。數字是**每 100**；讀不到的是 ""。
 *  `name` 只有 AI 會給（手機辨識不讀名稱，規格 §1.2）。 */
export type LabelFill = {
	baseUnit: BaseUnit;
	values: Record<Nutrient, string>;
	serving: string | null;
	name: string | null;
};

const ORDER: readonly Nutrient[] = ["kcal", "protein_g", "fat_g", "carb_g"];
const NAMES: Record<Nutrient, string> = {
	kcal: "熱量",
	protein_g: "蛋白質",
	fat_g: "脂肪",
	carb_g: "碳水化合物",
};
const UNITS: Record<Nutrient, string> = {
	kcal: "kcal",
	protein_g: "g",
	fat_g: "g",
	carb_g: "g",
};

export function fillFromAi(result: LabelReadResponse): LabelFill {
	const { nutrition } = result;
	return {
		baseUnit: nutrition.base_unit,
		values: {
			kcal: nutrition.kcal,
			protein_g: nutrition.protein_g,
			fat_g: nutrition.fat_g,
			carb_g: nutrition.carb_g,
		},
		serving: result.serving?.amount ?? null,
		name: result.name,
	};
}

/** 完整的與不完整的都用 `found`（完整時跟 `nutrition` 是同一組數字）。 */
export function fillFromParse(parse: LabelTextParse): LabelFill {
	const { found } = parse;
	return {
		baseUnit: parse.base_unit,
		values: {
			kcal: found.kcal ?? "",
			protein_g: found.protein_g ?? "",
			fat_g: found.fat_g ?? "",
			carb_g: found.carb_g ?? "",
		},
		serving: parse.serving?.amount ?? null,
		name: null,
	};
}

export function hasFoundValues(parse: LabelTextParse): boolean {
	return ORDER.some((nutrient) => parse.found[nutrient] !== null);
}

/** 讀不完整時怎麼說（規格 §3.1；`no_values` 的說法見計畫「與規格的差異」第 1 點）。 */
export function describeLabelParse(parse: LabelTextParse): string {
	switch (parse.reason) {
		case "needs_serving_size":
			return "只有每份的數字，但讀不到每一份量";
		case "columns_unknown":
			return "看不出是每份還是每 100 的數字";
		case "implausible":
			return "讀到的數字不太合理（可能少了小數點）";
		case "missing_fields": {
			const read: string[] = [];
			for (const nutrient of ORDER) {
				const value = parse.found[nutrient];
				if (value !== null) {
					read.push(`${NAMES[nutrient]} ${formatMacro(value)} ${UNITS[nutrient]}`);
				}
			}
			const missing = parse.missing.map((nutrient) => NAMES[nutrient]);
			return `讀到：${read.join("、")}（每 100 ${parse.base_unit}）；讀不到：${missing.join("、")}`;
		}
		default:
			return "讀不出營養標示的數字";
	}
}
```

```ts
import { shrinkToLongestEdge } from "./resize-image";

/** 手機辨識前縮圖的長邊（標示 OCR 規格 D11）：4000 px 的原圖慢一倍、還讀錯字；比這小的不放大。 */
export const OCR_MAX_EDGE = 2000;

/** 手機不能辨識：引擎載不起來、逾時、裝置不支援、縮圖失敗。畫面說這一句、給 AI 按鈕。 */
export class OcrUnavailableError extends Error {
	constructor(cause?: unknown) {
		super("這台裝置現在不能用手機辨識", { cause });
		this.name = "OcrUnavailableError";
	}
}

export function isAbortError(error: unknown): boolean {
	return error instanceof DOMException && error.name === "AbortError";
}

/** 營養標示的照片 → 辨識出來的字（可能是空字串）。**照片不離開裝置**（規格 §7）。
 *
 *  引擎（Tesseract.js、wasm、語言資料）只在這裡 `import()`：主程式不付這個成本（規格 D18）。
 *  中止（`signal`）丟 `AbortError`；其他失敗一律包成 `OcrUnavailableError`。 */
export async function recognizeLabel(
	file: File,
	signal: AbortSignal,
): Promise<string> {
	try {
		const image = await shrinkToLongestEdge(file, OCR_MAX_EDGE);
		signal.throwIfAborted();
		const engine = await import("./label-ocr-engine");
		signal.throwIfAborted();
		return await engine.recognizeText(image, signal);
	} catch (error) {
		if (isAbortError(error) || error instanceof OcrUnavailableError) throw error;
		throw new OcrUnavailableError(error);
	}
}
```

```ts
import corePlainUrl from "tesseract.js-core/tesseract-core-lstm.wasm.js?url";
import coreSimdUrl from "tesseract.js-core/tesseract-core-simd-lstm.wasm.js?url";
import workerUrl from "tesseract.js/dist/worker.min.js?url";
import { simd } from "wasm-feature-detect";

/** 手機辨識的引擎（標示 OCR 規格 §6）。**只能用 `import()` 載入**（`lib/label-ocr.ts`）。
 *
 *  三樣東西都從自己的網域來（規格 D12）：worker 與 core 用 `?url`（雜湊檔名，同 zxing 的 wasm），
 *  語言資料在 `vite.config.ts` 的 plugin 放的固定路徑。Tesseract.js 預設會去 jsDelivr 抓，所以
 *  `workerPath`、`corePath`、`langPath` 三個都要給。`workerBlobURL: false`：同源的 worker 直接用
 *  網址，不用 blob 包一層。
 *
 *  core 用單檔版（wasm 以 base64 包在 `.wasm.js` 裡）：分開的 `.wasm` 要靠固定的相對路徑找，跟
 *  雜湊檔名合不來。有 SIMD 用 `simd-lstm`，沒有用 `lstm`（一台裝置只會下載其中一個）。 */

/** 跟 `vite.config.ts` 的 `OCR_LANG_DIR` 同一個（前面多一個 `/`）。 */

export const OCR_TIMEOUT_MS = 90_000;

/** 圖片 → 字。PSM 6（單一區塊）＋自動轉正（規格 D11）。每次建一個 worker、用完收掉（D14）。
 *
 *  **Tesseract.js 7.0.0 的毛病（規格 D16、寫計畫時實測）**：語言資料抓不到、初始化失敗時，
 *  `createWorker` 的 promise 永遠不 settle，只有 `errorHandler` 會被叫——所以自己跟它、逾時、
 *  中止三個賽跑。失敗時拿不到那個 worker（計畫「與規格的差異」第 3 點）。 */
export async function recognizeText(
	image: Blob,
	signal: AbortSignal,
): Promise<string> {
	signal.throwIfAborted();
	let timer = 0;
	let reportFailure: (error: unknown) => void = () => {};
	const stopped = new Promise<never>((_, reject) => {
		timer = window.setTimeout(
			() => reject(new OcrUnavailableError("逾時")),
			OCR_TIMEOUT_MS,
		);
		signal.addEventListener("abort", () => reject(signal.reason), {
			once: true,
		});
		reportFailure = (error) => reject(new OcrUnavailableError(error));
	});
	// 沒有人在等的時候 reject 也不要變成 unhandled rejection。
	stopped.catch(() => undefined);

	const creating = createWorker("chi_tra", OEM.LSTM_ONLY, {
		workerPath: workerUrl,
		corePath: (await simd()) ? coreSimdUrl : corePlainUrl,
		langPath: new URL(OCR_LANG_PATH, window.location.origin).href,
		cachePath: OCR_CACHE_PATH,
		workerBlobURL: false,
		errorHandler: (error: unknown) => reportFailure(error),
	});
	let worker: Worker | null = null;
	try {
		worker = await Promise.race([creating, stopped]);
		await Promise.race([
			worker.setParameters({
				tessedit_pageseg_mode: PSM.SINGLE_BLOCK,
				preserve_interword_spaces: "1",
			}),
			stopped,
		]);
		const { data } = await Promise.race([
			worker.recognize(image, { rotateAuto: true }),
			stopped,
		]);
		return data.text;
	} finally {
		window.clearTimeout(timer);
		if (worker !== null) {
			await worker.terminate();
		} else {
			// 中止或逾時時 worker 可能還在建：建好了就收掉，不留一個在背景。
			creating.then(
				(late) => late.terminate(),
				() => undefined,
			);
		}
	}
}
```

```tsx
				show(fillFromAi(result), {
					source: "ai",
					partial: false,

				});
			},
		});
	}

	function fillPartial(parse: LabelTextParse, source: "ocr" | "paste") {
		show(fillFromParse(parse), appliedFromParse(source, parse, true));
	}

	function cancelCard() {
		stopOcr();
		setPending(null);
		setOcr(null);
		setPickError(null);
		readLabelMutation.reset();
		focusNextRef.current = "picker";
	}

	function reopenForAi() {
		if (lastPhoto === null) return;
		readLabelMutation.reset();
		setPending(lastPhoto);
		setOcr({ status: "done", message: "用同一張照片改用 AI 讀", partial: null });
		focusNextRef.current = "step";
	}

	function submitPaste() {
		const text = pasteText.trim();
		if (text === "" || pasteMutation.isPending) return;
		setPasteMessage(null);
		setPastePartial(null);
		pasteMutation.mutate(text, {
			onSuccess: (parse) => {
				if (parse.complete) {
					setPasteOpen(false);
					setPasteText("");
					show(fillFromParse(parse), appliedFromParse("paste", parse, false));
					return;
				}
				setPasteMessage(describeLabelParse(parse));
				setPastePartial(hasFoundValues(parse) ? parse : null);
			},
			onError: (error) => setPasteMessage(describeLabelTextError(error)),
		});
	}

	function closePaste() {
		setPasteOpen(false);
		setPasteMessage(null);
		setPastePartial(null);
	}

	const aiPending = readLabelMutation.isPending;
	const pasteBlocked = pasteMutation.isPending || pasteText.trim() === "";

	return (
		<>
			<div className={styles.labelRow}>
				{/* 要拍，所以帶 capture（掃描層的「從照片讀條碼」不帶——那一顆要能選相簿）。 */}
				<PhotoPickerButton
					id="label-photo"
					label="拍營養標示"
					accept="image/*"
					capture="environment"
					variant="accent"
					disabled={busy}
					inputRef={inputRef}
					onChange={handlePick}
				/>
				<button
					type="button"
					className={base.aiButton}
					aria-expanded={pasteOpen}
					aria-controls={pasteId}
					onClick={() => (pasteOpen ? closePaste() : setPasteOpen(true))}
				>
					貼上營養標示文字
				</button>
			</div>
			{pickError !== null && <p role="alert">{pickError}</p>}
			{pending !== null && (
				<section
					ref={stepRef}
					tabIndex={-1}
					aria-label="確認照片"
					className={`${base.card} ${step.step}`}
				>
					<LabelPreview file={pending.file} />
					<p role="status" className={styles.ocrMessage}>
						{ocr?.status === "done" ? ocr.message : ""}
					</p>
					{ocr?.status === "done" && (
						<HintField
							hint={pending.hint}
							onHintChange={(hint) => setPending({ ...pending, hint })}
							placeholder="例：這是每 100 ml、一包 2 份"
							readOnly={aiPending}
						/>
					)}
					<div className={base.actions}>
						{ocr?.status === "done" && (
							<button
								type="button"
								className={base.primary}
								aria-disabled={aiPending}
								onClick={() => {
									if (!aiPending) startAi(pending);
								}}
							>
								{AI_LABEL}
							</button>
						)}
						{ocr?.status === "done" && ocr.partial !== null && (
							<button
								type="button"
								className={base.secondary}
								aria-disabled={aiPending}
								onClick={() => {
									if (aiPending || ocr.partial === null) return;
									const partial = ocr.partial;

									setOcr(null);
									setLastPhoto(pending);
									fillPartial(partial, "ocr");

							>
								填入讀到的
							</button>
						)}
						<button
							type="button"
							className={base.secondary}
							aria-disabled={aiPending}
							onClick={() => {
								if (!aiPending) cancelCard();
							}}
						>
							取消
						</button>
					</div>
				</section>
			)}
			{pasteOpen && (
				<section
					id={pasteId}
					aria-label="貼上營養標示文字"

						readOnly={pasteMutation.isPending}
						onChange={(event) => setPasteText(event.target.value)}
					/>
					<p role="status">{pasteMessage ?? ""}</p>
					<div className={base.actions}>
						<button
							type="button"
							className={base.primary}
							aria-disabled={pasteBlocked}
							onClick={submitPaste}
						>
							讀取文字
						</button>
						{pastePartial !== null && (
							<button
								type="button"
								className={base.secondary}
								onClick={() => {
									const partial = pastePartial;
									closePaste();
									setPasteText("");
									fillPartial(partial, "paste");
								}}
							>
								填入讀到的
							</button>

						<button
							type="button"
							className={base.secondary}
							onClick={closePaste}

				<div className={styles.statusRow}>
					<div role="status">
						{ocrRunning && (

									{applied.partial ? PARTIAL_NOTE : NOTES[applied.source]}
								</p>
								{applied.flagged && (
									<p className={styles.warning}>
										熱量跟三大營養素算起來對不太上，請特別確認

										今天還能用 {applied.remainingToday} 次

						)}
					</div>
					{busy && <ElapsedSeconds className={styles.muted} />}
				</div>
				{applied?.source === "ocr" &&
					!busy &&
					pending === null &&
					lastPhoto !== null && (
						<button
							type="button"
							className={base.aiButton}
							onClick={reopenForAi}
						>
							{applied.partial ? AI_LABEL : WRONG_LABEL}
						</button>
					)}
				{readLabelMutation.isError && (
					<p role="alert">{describeAnalyzeError(readLabelMutation.error)}</p>
				)}
			</div>
		</>
	);
}
```
